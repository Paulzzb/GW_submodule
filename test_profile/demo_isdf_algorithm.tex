\section{Notation}
For each $\qq$-point in the IBZ:
\begin{itemize}
  \item $F$: FFT from real-space grid to $G$-components (Coulomb basis).
  \item $M_{\qq} \equiv \mathrm{MCH}_{\qq}$: accumulated fine-grid coupling matrix ($n_c \times N_\mu$).
  \item $C_{\qq} \equiv \mathrm{CCH}_{\qq}$: accumulated ISDF overlap matrix ($N_\mu \times N_\mu$), symmetric PSD.
  \item $V_{\qq}$: Coulomb operator, diagonal in $G$-space.
  \item $c \in \mathbb{C}^{N_\mu}$: ISDF coefficient vector (e.g.\ density $\rho$ on sampling points, from \texttt{get\_u\_xalpha} / \texttt{get\_rho\_xalpha}).
\end{itemize}

\section{Current algorithm (\texttt{isdf.gen\_tildeVq})}

\subsection{Type~a (symmetrized form)}
\begin{align}
  G_{\qq}^{(1)} &= (F M_{\qq})^{\mathsf{H}} V_{\qq} (F M_{\qq}), \\
  \tilde{V}_{\qq}^{(2)} &= C_{\qq}^{-1}\, G_{\qq}^{(1)}\, C_{\qq}^{-1}.
\end{align}

\subsection{Type~b (helper form, implemented)}
\begin{align}
  P_{\qq} &= M_{\qq}\, C_{\qq}^{-1}, \qquad
  \tilde{H}_{\qq} = F P_{\qq}, \\
  \tilde{V}_{\qq} &= \tilde{H}_{\qq}^{\mathsf{H}} V_{\qq} \tilde{H}_{\qq}
  = (F M_{\qq} C_{\qq}^{-1})^{\mathsf{H}} V_{\qq} (F M_{\qq} C_{\qq}^{-1}).
\end{align}
In code: \texttt{helperqR = MCHq / CCHq}, FFT $\to$ \texttt{helperqG}, then
\texttt{tildeVq = helperqG' * diag(vcoul) * helperqG}.

\subsection{Application (current)}
\begin{equation}
  \mathcal{E} = c^{\mathsf{H}} \tilde{V}_{\qq}\, c.
\end{equation}

\subsection{Inverting $C_{\qq}$ (three options under study)}
\begin{enumerate}
  \item \textbf{Direct / Cholesky:} $C_{\qq} = L L^{\mathsf{H}}$, use $C_{\qq}^{-1} = L^{-\mathsf{H}} L^{-1}$.
  \item \textbf{Truncated pseudoinverse (SVD):}
  $C_{\qq} = \sum_i \sigma_i v_i v_i^{\mathsf{H}}$, keep modes with $\sigma_i/\sigma_1 > s$:
  \[
    C_{\qq}^{-1} \approx \sum_{i < \ell} \frac{1}{\sigma_i}\, v_i v_i^{\mathsf{H}}.
  \]
  \item \textbf{Shifted Cholesky:} factor $(C_{\qq} + s I) = L L^{\mathsf{H}}$, approximate $C_{\qq}^{-1}$.
\end{enumerate}
Goal: quantify how truncation / shift in (2)--(3) affects $\tilde{V}_{\qq}$ and downstream energies.

\section{Proposed SVD test algorithm (\texttt{isdftest.gen\_tildeVq})}

Avoid explicit $C_{\qq}^{-1}$; use the matrix square root from SVD:
\begin{equation}
  C_{\qq} = \sum_i \sigma_i v_i v_i^{\mathsf{H}}
  \quad\Longrightarrow\quad
  C_{\qq}^{1/2} = \sum_i \sqrt{\sigma_i}\, v_i v_i^{\mathsf{H}}.
\end{equation}
where $\{\sigma_i\}$ is in descending order, and $\sigma_{l+1} < s\sigma_1 < \sigma_l$.

\subsection{Construction}
\begin{align}
  P_{\qq} &= M_{\qq}\, \bigl(\sum_{i=1}^l \frac{1}{\sqrt{\sigma_i}} v_i v_i\herm \bigr), \\
  \tilde{H}_{\qq} &= F P_{\qq}, \\
  \tilde{V}_{\qq}^{\mathrm{new}} &= \tilde{H}_{\qq}^{\mathsf{H}} V_{\qq} \tilde{H}_{\qq}.
\end{align}

\subsection{Application (proposed)}
Transform coefficients before contraction:
\begin{align}
  c_t &=  \bigl(\sum_{i=1}^l \frac{1}{\sqrt{\sigma_i}} v_i v_i\herm \bigr) c,\\
  \mathcal{E}^{\mathrm{new}} &= c_t^{\mathsf{H}} \tilde{V}_{\qq}^{\mathrm{new}}\, c_t.
\end{align}
In \texttt{+isdftest}: \texttt{get\_rho\_xalpha} returns $c_t = C_{\qq}^{-1/2} c$;
contraction is $c_t^{\mathsf{H}} \tilde{V}_{\qq}^{\mathrm{new}}\, c_t$ as in \texttt{+isdf}.
$C_{\qq}^{-1/2}$ is cached per $\qq$ during \texttt{gen\_tildeVq}.
